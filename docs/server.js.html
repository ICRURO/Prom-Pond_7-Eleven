const express = require('express');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = process.env.PORT || 3000;
const DB_PATH = path.join(__dirname, 'data', 'database.json');

app.use(express.json());
app.use(express.static(path.join(__dirname, 'public')));

/**
 * Lee y parsea el archivo de base de datos local JSON.
 * @returns {BaseDeDatos} Objeto con las colecciones de estudiantes, experiencias educativas e inscripciones.
 */
const readDB = () => {
    try {
        const raw = fs.readFileSync(DB_PATH, 'utf-8');
        return JSON.parse(raw);
    } catch (error) {
        return { estudiantes: [], experiencias_educativas: [], inscripciones: [] };
    }
};

/**
 * Sobrescribe el archivo de base de datos local JSON con la información actualizada.
 * @param {BaseDeDatos} data - Estructura completa de la base de datos a persistir.
 * @returns {boolean} Verdadero si la escritura fue exitosa, falso en caso de error.
 */
const writeDB = (data) => {
    try {
        fs.writeFileSync(DB_PATH, JSON.stringify(data, null, 2), 'utf-8');
        return true;
    } catch (err) {
        console.error("Error escribiendo DB:", err);
        return false;
    }
};

/**
 * Obtiene la lista completa de estudiantes registrados.
 * @name GET /api/estudiantes
 * @function
 * @param {express.Request} req - Objeto de petición HTTP.
 * @param {express.Response} res - Objeto de respuesta HTTP.
 * @returns {Estudiante[]} 200 - Arreglo con todos los estudiantes.
 */
app.get('/api/estudiantes', (req, res) => {
    const db = readDB();
    res.json(db.estudiantes || []);
});

/**
 * Registra un nuevo estudiante en el sistema.
 * @name POST /api/estudiantes
 * @function
 * @param {express.Request} req - Objeto de petición con matricula, nombre, carrera y semestre en el cuerpo.
 * @param {express.Response} res - Objeto de respuesta HTTP.
 * @returns {Estudiante} 201 - Estudiante recién creado.
 * @returns {Object} 400 - Mensaje de error si faltan campos o si la matrícula ya existe.
 */
app.post('/api/estudiantes', (req, res) => {
    const { matricula, nombre, carrera, semestre } = req.body;
    const db = readDB();
    if (!matricula || !nombre) {
        return res.status(400).json({ error: 'Matrícula y nombre son requeridos.' });
    }
    if (db.estudiantes.some(e => e.matricula.trim().toUpperCase() === matricula.trim().toUpperCase())) {
        return res.status(400).json({ error: 'La matrícula ya existe.' });
    }
    const nuevo = {
        matricula: matricula.trim().toUpperCase(),
        nombre: nombre.trim(),
        carrera: carrera.trim(),
        semestre: (semestre || '').trim()
    };
    db.estudiantes.push(nuevo);
    writeDB(db);
    res.status(201).json(nuevo);
});

/**
 * Actualiza los datos de un estudiante identificado por su matrícula.
 * @name PUT /api/estudiantes/:matricula
 * @function
 * @param {express.Request} req - Petición con parámetro :matricula y campos a modificar en el cuerpo.
 * @param {express.Response} res - Objeto de respuesta HTTP.
 * @returns {Estudiante} 200 - Objeto del estudiante con los datos actualizados.
 * @returns {Object} 404 - Mensaje de error si no se localiza la matrícula.
 */
app.put('/api/estudiantes/:matricula', (req, res) => {
    const matriculaParam = req.params.matricula.trim().toUpperCase();
    const { nombre, carrera, semestre } = req.body;
    const db = readDB();
    const idx = db.estudiantes.findIndex(e => e.matricula.trim().toUpperCase() === matriculaParam);
    if (idx === -1) return res.status(404).json({ error: 'Estudiante no encontrado.' });
    db.estudiantes[idx].nombre = nombre !== undefined ? nombre.trim() : db.estudiantes[idx].nombre;
    db.estudiantes[idx].carrera = carrera !== undefined ? carrera.trim() : db.estudiantes[idx].carrera;
    if (semestre !== undefined && semestre !== "") {
        db.estudiantes[idx].semestre = semestre.trim();
    }
    writeDB(db);
    res.json(db.estudiantes[idx]);
});

/**
 * Elimina un estudiante y todas sus inscripciones asociadas.
 * @name DELETE /api/estudiantes/:matricula
 * @function
 * @param {express.Request} req - Petición con parámetro :matricula en la URL.
 * @param {express.Response} res - Objeto de respuesta HTTP.
 * @returns {Object} 200 - Bandera { success: true }.
 */
app.delete('/api/estudiantes/:matricula', (req, res) => {
    const matriculaParam = req.params.matricula.trim().toUpperCase();
    let db = readDB();
    db.estudiantes = db.estudiantes.filter(e => e.matricula.trim().toUpperCase() !== matriculaParam);
    db.inscripciones = db.inscripciones.filter(i => i.matricula.trim().toUpperCase() !== matriculaParam);
    writeDB(db);
    res.json({ success: true });
});

app.get('/api/materias', (req, res) => {

/**
 * Agrega una nueva experiencia educativa al catálogo.
 * @name POST /api/materias
 * @function
 * @param {express.Request} req - Petición con nrc, nombre, carrera y creditos en el cuerpo.
 * @param {express.Response} res - Objeto de respuesta HTTP.
 * @returns {ExperienciaEducativa} 201 - Experiencia educativa agregada.
 * @returns {Object} 400 - Mensaje de error si faltan datos o el NRC ya existe.
 */
app.post('/api/materias', (req, res) => {
    const { nrc, nombre, carrera, creditos } = req.body;
    const db = readDB();
    const nrcLimpio = String(nrc).trim();
    if (!nrcLimpio || !nombre || creditos === undefined) {
        return res.status(400).json({ error: 'NRC, nombre y créditos son requeridos.' });
    }
    if (db.experiencias_educativas.some(e => String(e.nrc) === nrcLimpio)) {
        return res.status(400).json({ error: 'El NRC ya existe en el catálogo.' });
    }
    const nuevaEE = {
        nrc: nrcLimpio,
        nombre: nombre.trim(),
        carrera: carrera ? carrera.trim().toUpperCase() : 'GENERAL',
        creditos: parseFloat(creditos)
    };
    db.experiencias_educativas.push(nuevaEE);
    writeDB(db);
    res.status(201).json(nuevaEE);
});

/**
 * Modifica los datos de una experiencia educativa existente mediante su NRC.
 * @name PUT /api/materias/:nrc
 * @function
 * @param {express.Request} req - Petición con el parámetro :nrc y campos actualizados en el cuerpo.
 * @param {express.Response} res - Objeto de respuesta HTTP.
 * @returns {Object} 200 - Objeto con confirmación de éxito y la materia editada.
 * @returns {Object} 404 - Mensaje de error si no existe el NRC.
 */
app.put('/api/materias/:nrc', (req, res) => {
    const nrcParam = String(req.params.nrc).trim();
    const { nombre, carrera, creditos } = req.body;
    const db = readDB();
    const ee = db.experiencias_educativas.find(e => String(e.nrc) === nrcParam);
    if (!ee) return res.status(404).json({ error: 'Experiencia Educativa no encontrada.' });
    if (nombre) ee.nombre = nombre.trim();
    if (carrera) ee.carrera = carrera.trim().toUpperCase();
    if (creditos !== undefined) ee.creditos = parseFloat(creditos);
    writeDB(db);
    res.json({ success: true, ee });
});

/**
 * Elimina una experiencia educativa del catálogo y remueve sus inscripciones vinculadas.
 * @name DELETE /api/materias/:nrc
 * @function
 * @param {express.Request} req - Petición con parámetro :nrc en la URL.
 * @param {express.Response} res - Objeto de respuesta HTTP.
 * @returns {Object} 200 - Bandera { success: true }.
 */
app.delete('/api/materias/:nrc', (req, res) => {
    const nrcParam = String(req.params.nrc).trim();
    let db = readDB();
    db.experiencias_educativas = db.experiencias_educativas.filter(e => String(e.nrc) !== nrcParam);
    db.inscripciones = db.inscripciones.filter(i => String(i.nrc) !== nrcParam);
    writeDB(db);
    res.json({ success: true });
});

/**
 * Consulta el kardex completo de un estudiante (datos personales y materias cursadas).
 * @name GET /api/estudiantes/:matricula/kardex
 * @function
 * @param {express.Request} req - Petición con la matrícula del alumno en :matricula.
 * @param {express.Response} res - Objeto de respuesta HTTP.
 * @returns {Object} 200 - Objeto con datos del estudiante y el arreglo detallado de sus materias.
 * @returns {Object} 404 - Mensaje de error si el estudiante no existe.
 */
app.get('/api/estudiantes/:matricula/kardex', (req, res) => {
    const matriculaParam = req.params.matricula.trim().toUpperCase();
    const db = readDB();
    const estudiante = db.estudiantes.find(e => e.matricula.trim().toUpperCase() === matriculaParam);
    if (!estudiante) return res.status(404).json({ error: 'Estudiante no encontrado.' });
    const inscripcionesAlumno = db.inscripciones.filter(i => i.matricula.trim().toUpperCase() === matriculaParam);
    const materias = inscripcionesAlumno.map(ins => {
        const ee = db.experiencias_educativas.find(e => String(e.nrc) === String(ins.nrc));
        return {
            id_inscripcion: String(ins.id),
            nrc: ins.nrc,
            nombre: ee ? ee.nombre : 'Materia desconocida',
            carrera: ee ? ee.carrera : 'N/A',
            creditos: ee ? ee.creditos : 0,
            calificacion: ins.calificacion
        };
    });
    res.json({ estudiante, materias });
});

/**
 * Inscribe una experiencia educativa al kardex de un estudiante con su calificación obtenida.
 * @name POST /api/estudiantes/:matricula/kardex
 * @function
 * @param {express.Request} req - Petición con matrícula en la URL y cuerpo con { nrc, calificacion }.
 * @param {express.Response} res - Objeto de respuesta HTTP.
 * @returns {Inscripcion} 201 - Registro de inscripción generado.
 * @returns {Object} 400 - Mensaje si la materia ya está inscrita.
 * @returns {Object} 404 - Mensaje si la materia no existe en el catálogo.
 */
app.post('/api/estudiantes/:matricula/kardex', (req, res) => {
    const matriculaParam = req.params.matricula.trim().toUpperCase();
    const { nrc, calificacion } = req.body;
    const db = readDB();
    const nrcLimpio = String(nrc).trim();
    const ee = db.experiencias_educativas.find(e => String(e.nrc) === nrcLimpio);
    if (!ee) {
        return res.status(404).json({ error: 'La materia seleccionada no existe en el catálogo.' });
    }
    const yaInscrita = db.inscripciones.some(
        i => i.matricula.trim().toUpperCase() === matriculaParam && String(i.nrc) === nrcLimpio
    );
    if (yaInscrita) {
        return res.status(400).json({ error: 'El estudiante ya tiene cursada o inscrita esta materia.' });
    }
    const nuevaInscripcion = {
        id: String(Date.now()),
        matricula: matriculaParam,
        nrc: nrcLimpio,
        calificacion: calificacion.toString().toUpperCase() === 'AC' ? 'AC' : parseFloat(calificacion)
    };
    db.inscripciones.push(nuevaInscripcion);
    writeDB(db);
    res.status(201).json(nuevaInscripcion);
});

/**
 * Da de baja o elimina una inscripción específica del historial académico.
 * @name DELETE /api/inscripciones/:id
 * @function
 * @param {express.Request} req - Petición con el identificador único de inscripción en :id.
 * @param {express.Response} res - Objeto de respuesta HTTP.
 * @returns {Object} 200 - Bandera { success: true }.
 */
app.delete('/api/inscripciones/:id', (req, res) => {
    const idParam = String(req.params.id);
    let db = readDB();
    db.inscripciones = db.inscripciones.filter(i => String(i.id) !== idParam);
    writeDB(db);
    res.json({ success: true });
});
